# FASE 3.3: Contrato do DisputeResolver (Core Isolado)

Este é o terceiro módulo do Core. Deve residir em `src/core/dispute/DisputeResolver.ts`.
**Zero dependências externas.** Apenas TypeScript nativo.
**Responsabilidade única:** Gerenciar a máquina de estados da disputa e validar votos. Não acessa banco, não notifica, não libera pagamentos.

---

## 1. 📐 Tipos e Interfaces
*Local: `src/core/dispute/types.ts`*

```typescript
/**
 * Status possíveis de uma disputa (máquina de estados).
 */
export type DisputeStatus = 'pending' | 'resolved' | 'escalated' | 'expired';

/**
 * Opções de voto do comitê.
 */
export type VoteOption = 'approve' | 'reject';

/**
 * Papéis dos membros do comitê de arbitragem.
 */
export type CommitteeRole = 'platform' | 'external_maintainer' | 'community_lancer';

/**
 * Membro do comitê de arbitragem.
 */
export interface CommitteeMember {
  userId: string;
  role: CommitteeRole;
}

/**
 * Voto individual registrado no comitê.
 */
export interface DisputeVote {
  voterId: string;
  role: CommitteeRole;
  option: VoteOption;
  justification: string;      // Texto obrigatório (mínimo 50 caracteres)
  votedAt: string;            // ISO 8601
}

/**
 * Dados de entrada para criação de uma disputa.
 */
export interface DisputeInput {
  disputeId: string;
  challengerId: string;       // Lancer que contestou
  challengedId: string;       // Mantenedor avaliado
  challengeId: string;        // Desafio original
  prUrl: string;              // PR em disputa
  originalScore: number;      // Score original calculado
  contestedScore: number;     // Score alegado pelo challenger
  reason: string;             // Motivo da contestação
  createdAt: string;          // ISO 8601
  deadlineAt: string;         // createdAt + 7 dias
  committee: CommitteeMember[]; // Deve ter exatamente 3 membros
}

/**
 * Estado completo da disputa (imutável após cada operação).
 */
export interface Dispute {
  disputeId: string;
  status: DisputeStatus;
  votes: DisputeVote[];
  decision: VoteOption | null;  // null se não resolvida
  resolvedAt: string | null;
  escalatedAt: string | null;
  createdAt: string;
  deadlineAt: string;
}

/**
 * Resultado de uma operação no DisputeResolver.
 * O Core NUNCA lança exceções; retorna este objeto.
 */
export type DisputeResult = 
  | { success: true; dispute: Dispute }
  | { success: false; error: DisputeError };

/**
 * Erros possíveis nas operações de disputa.
 */
export interface DisputeError {
  code: DisputeErrorCode;
  field?: string;
  message: string;
}

export type DisputeErrorCode =
  | 'ALREADY_RESOLVED'
  | 'ALREADY_ESCALATED'
  | 'ALREADY_EXPIRED'
  | 'INVALID_REFERENCE'
  | 'INVALID_VOTER'
  | 'DUPLICATE_VOTE'
  | 'CONFLICT_OF_INTEREST'
  | 'INVALID_JUSTIFICATION'
  | 'INVALID_COMMITTEE'
  | 'INVALID_DEADLINE'
  | 'INVALID_INPUT';
