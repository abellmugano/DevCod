## 1. 📐 Tipos e Interfaces
*Local: `src/core/score/types.ts`*

```typescript
/**
 * Métricas automáticas coletadas via CI/CD (metrics.json).
 * Se ausentes, o ScoreCalculator aplica penalidade de 20%.
 */
export interface MetricsData {
  domainFilesChanged: number;        // Arquivos de domínio alterados (0-N)
  cyclomaticComplexityDelta: number; // Variação da complexidade ciclomática (negativo = melhoria)
  assertionsAdded: number;           // Número de asserções de teste adicionadas (0-N)
  diffCoverage: number;              // Cobertura de teste das linhas alteradas (0-100)
  lintPassed: boolean;               // Status do linting/type-checking
}

/**
 * Avaliação manual do mantenedor (escala 1-5 para cada critério).
 */
export interface ManualEvaluation {
  roadmapImpact: number;    // Impacto no roadmap (1-5)
  codeQuality: number;      // Qualidade do código na revisão (1-5)
  documentationClarity: number; // Clareza da documentação (1-5)
}

/**
 * Dados de entrada para cálculo do Score.
 */
export interface ScoreInput {
  metrics: MetricsData | null;  // null se metrics.json ausente
  manualEvaluation: ManualEvaluation;
  algorithmVersion: string;     // Ex: "score_v1"
}

/**
 * Resultado do cálculo do Score.
 * Score final é sempre 0-10.
 */
export interface ScoreResult {
  totalScore: number;           // Score final (0-10)
  autoScore: number;            // Componente automático (0-10)
  manualScore: number;          // Componente manual (0-10)
  metricsPresent: boolean;      // true se metrics.json foi fornecido
  penaltyApplied: boolean;      // true se penalidade de 20% foi aplicada
  algorithmVersion: string;     // Versão do algoritmo usado
  breakdown: ScoreBreakdown;    // Detalhamento para auditoria
}

/**
 * Detalhamento do cálculo (para auditoria forense no event_log).
 */
export interface ScoreBreakdown {
  domainFilesScore: number;     // 0-10
  complexityScore: number;      // 0-10
  assertionsScore: number;      // 0-10
  coverageScore: number;        // 0-10
  lintScore: number;            // 0-10
  roadmapImpactScore: number;   // 0-10
  codeQualityScore: number;     // 0-10
  documentationScore: number;   // 0-10
}

/**
 * Erro de cálculo (ex: avaliação manual fora do range).
 */
export interface ScoreCalculationError {
  field: string;
  code: 'INVALID_RANGE' | 'MISSING_DATA';
  message: string;
}
